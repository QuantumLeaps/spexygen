  % Customer-specific license line shown at the bottom of the title page.
  % $CUST_NAME$/$CUST_URL$ are placeholders -- substitute with the actual
  % customer name/URL per delivered copy.
  %
  % The literal "$" and "_" characters in these placeholders are TeX special
  % characters (math-mode delimiter and subscript, respectively) and must be
  % escaped in the visible link text (\$...\$, \_) to print literally rather
  % than tripping into math mode -- confirmed by testing: unescaped, "$CUST_
  % NAME$" was silently typeset as math with NAME as a subscript of CUST.
  % The URL argument of \href is parsed specially by url.sty (verbatim-like
  % catcodes) and must stay unescaped -- adding the same backslash-escapes
  % there causes a runaway "TeX capacity exceeded" failure, confirmed by
  % testing.
  \newcommand{\licenseinfo}{Licensed to \href{$CUST_URL$}{\$CUST\_NAME\$}}

  % Footer layout mirrors the header's inner/outer split: $projectnumber
  % moves to the outer edge (LE/RO), directly below where \thepage sits in
  % the header above it, and \licenseinfo (defined earlier alongside
  % \contactdetails/title page) takes over the inner edge (LO/RE) that used
  % to repeat \thepage -- the page number already appears once, in the
  % header, so it's not duplicated here. Footer text originally matched the
  % header's font/size exactly (\bfseries at normal size); dialed back one
  % step to \small so the footer reads a tad smaller than the header while
  % still bold. Both header and footer now also pick up \sffamily, matching
  % the section-title font (\doxysection/etc. in spexy-style.sty use
  % \sffamily\bfseries) rather than the body's serif face.
  \fancypagestyle{fancyplain}{
    \fancyhf{}
    \fancyhead[LE, RO]{\sffamily\bfseries\thepage}
    \fancyhead[LO]{\sffamily\bfseries\rightmark}
    \fancyhead[RE]{\sffamily\bfseries\leftmark}
    \fancyfoot[LE, RO]{\sffamily\small\bfseries $projectnumber}
    \fancyfoot[LO, RE]{\sffamily\small\bfseries \licenseinfo}
  }
